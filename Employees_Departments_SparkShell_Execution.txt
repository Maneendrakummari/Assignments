4. COMMON TABLE EXPRESSIONS (CTEs)

Exercise 4.1 – Total payroll per department.
scala> spark.sql("""
     |   with dept_payroll as (
     |     select dept_id, sum(salary) as total_payroll
     |     from employees
     |     group by dept_id
     |   )
     |   select d.dept_name, dp.total_payroll
     |   from dept_payroll dp
     |   join departments d on dp.dept_id = d.dept_id
     |   order by dp.total_payroll desc
     | """).show()

Exercise 4.2 – Employees earning above company average.
scala> spark.sql("""
     |   with company_avg as (
     |     select avg(salary) as avg_sal
     |     from employees
     |   ),
     |   above_avg as (
     |     select e.emp_id, e.name, e.salary, c.avg_sal
     |     from employees e
     |     cross join company_avg c
     |     where e.salary > c.avg_sal
     |   )
     |   select name, salary,
     |          round(salary - avg_sal, 2) as above_by
     |   from above_avg
     |   order by above_by desc
     | """).show()

Exercise 5.1 – Employee details with department name and location.

Exercise 5.2 – All departments, including departments without employees.
scala> spark.sql("""
     |   select d.dept_id, d.dept_name, e.emp_id, e.name
     |   from departments d
     |   left join employees e on d.dept_id = e.dept_id

     | """).show()

Exercise 5.3 – Full outer join with an inline bonus-eligible set.
scala> spark.sql("""
     |   with bonus_eligible as (
     |     select 101 as emp_id, 'Y' as bonus_flag union all
     |     select 103, 'Y' union all
     |     select 108, 'Y' union all
     |     select 999, 'Y'
     |   )
     |   select coalesce(e.emp_id, b.emp_id) as emp_id,
     |          e.name,
     |          b.bonus_flag
     |   from employees e
     |   full outer join bonus_eligible b on e.emp_id = b.emp_id
     |   order by emp_id
     | """).show()

Exercise 5.4 – Employee and manager names using self join.

6. WINDOW FUNCTIONS
============================================================

Exercise 6.1 – row_number, rank and dense_rank within each department.
scala> spark.sql("""
     |   select name, dept_id, salary,
     |          row_number() over (partition by dept_id order by salary desc) as rn,
     |          rank() over (partition by dept_id order by salary desc) as rnk,
     |          dense_rank() over (partition by dept_id order by salary desc) as drnk
     |   from employees
     |   order by dept_id, salary desc
     | """).show()

Exercise 6.2 – Previous and next salary using lag and lead.
scala> spark.sql("""
     |   select dept_id, name, hire_date, salary,
     |          lag(salary) over (partition by dept_id order by hire_date) as prev_salary,
     |          lead(salary) over (partition by dept_id order by hire_date) as next_salary
     |   from employees
     |   order by dept_id, hire_date
     | """).show()

Exercise 6.3 – Running total of salary within each department.
scala> spark.sql("""
     |   select dept_id, name, hire_date, salary,
     |          sum(salary) over (
     |            partition by dept_id
     |            order by hire_date
     |            rows between unbounded preceding and current row
     |          ) as running_salary
     |   from employees
     |   order by dept_id, hire_date
     | """).show()

9. SQL VS DATAFRAME API
============================================================

Business problem – Find the highest-paid employee in each department.

Solution A – Pure Spark SQL.
scala> spark.sql("""
     |   with ranked as (
     |     select e.name, e.salary, d.dept_name,
     |            row_number() over (
     |              partition by e.dept_id
     |              order by e.salary desc
     |            ) as rn
     |     from employees e
     |     join departments d on e.dept_id = d.dept_id
     |   )
     |   select dept_name, name as top_employee, salary
     |   from ranked
     |   where rn = 1
     |   order by dept_name
     | """).show()

Solution B – DataFrame API (included for comparison).
scala> import org.apache.spark.sql.expressions.Window

scala> val w = Window.partitionBy("dept_id").orderBy(col("salary").desc)

scala> val result = employees
     |   .withColumn("rn", row_number().over(w))
     |   .filter(col("rn") === 1)
     |   .join(departments, "dept_id")
     |   .select(
     |     col("dept_name"),
     |     col("name").as("top_employee"),
     |     col("salary")
     |   )
     |   .orderBy("dept_name")

scala> result.show()

10. COMBINED CHALLENGE EXERCISES

Challenge 10.1 – Highest-paid employee in each location.
scala> spark.sql("""
     |   with ranked as (
     |     select d.location, e.name, e.salary,
     |            row_number() over (
     |              partition by d.location
     |              order by e.salary desc
     |            ) as rn
     |     from employees e
     |     join departments d on e.dept_id = d.dept_id
     |   )
     |   select location, name, salary
     |   from ranked
     |   where rn = 1
     |   order by location
     | """).show()

Challenge 10.2 – Skill count, whether employee has Spark, and skill names.
scala> spark.sql("""
     |   select emp_id, name,
     |          size(skills) as skill_count,
     |          exists(skills, x -> x.skill = 'Spark') as has_spark,
     |          transform(skills, x -> x.skill) as skill_names
     |   from emp_skills
     |   order by skill_count desc
     | """).show(false)

Challenge 10.3 – Employee and manager salary gap.
scala> spark.sql("""
     |   select e.name as employee,
     |          m.name as manager,
     |          e.salary as emp_salary,
     |          m.salary as mgr_salary,
     |          round(m.salary - e.salary, 2) as gap
     |   from employees e
     |   join employees m on e.manager_id = m.emp_id
     |   order by gap desc
     | """).show()

Challenge 10.4 – Exploded skills with running skill count per department.
scala> spark.sql("""
     |   with exploded as (
     |     select d.dept_name, e.name, s.skill, s.level
     |     from employees e
     |     join departments d on e.dept_id = d.dept_id
     |     lateral view explode(
     |       from_json(e.skills_json, 'array<struct<skill:string,level:string>>')
     |     ) t as s
     |   )
     |   select dept_name, name, skill, level,
     |          count(*) over (
     |            partition by dept_name
     |            order by name, skill
     |            rows between unbounded preceding and current row
     |          ) as running_skill_rows
     |   from exploded
     |   order by dept_name, name, skill
     | """).show()
