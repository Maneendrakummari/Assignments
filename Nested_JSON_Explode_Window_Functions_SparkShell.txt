     |   StructField("orderDate", StringType),

scala> val products = orders.select(
     |   col("order_id"),
     |   col("customer_id"),
     |   col("customer_name"),
     |   col("city"),
     |   col("order_date"),
     |   explode(col("items")).as("item"),
     |   col("payment_status")
     | ).select(
     |   col("order_id"),
     |   col("customer_id"),
     |   col("customer_name"),
     |   col("city"),
     |   col("order_date"),
     |   col("item.product").as("product"),
     |   col("item.quantity").as("quantity"),
     |   col("item.price").as("price"),
     |   (col("item.quantity") * col("item.price")).as("product_amount"),
     |   col("payment_status")
     | )

scala> val orderTotals = products.groupBy("order_id", "customer_id", "customer_name", "city", "order_date").agg(sum("product_amount").as("order_total"))
